\begin{frame}{Basic definitions: labelled transition systems}
\Margines{
\tiny
\begin{definition}
A \emph{labelled transition system} (LTS) is a tuple
\[
  \mathcal T=(C,\Sigma,\longrightarrow,c_0),
\]
where $C$ is a set of configurations, $\Sigma$ is a finite alphabet,
$c_0\in C$, and
\[
  \longrightarrow\ \subseteq C\times\Sigma\times C.
\]
We write $c\xrightarrow{a}c'$ for $(c,a,c')\in\longrightarrow$.
\end{definition}

An \emph{infinite run} is a sequence
\[
 c_0\xrightarrow{a_0}c_1\xrightarrow{a_1}c_2\xrightarrow{a_2}\cdots,
\]
and its \emph{trace} is the word $a_0a_1a_2\cdots\in\Sigma^\omega$.

\medskip
For an operational model $M$, its \emph{induced LTS} is obtained by taking
the configurations of $M$ as vertices and the one-step operational
semantics as labelled edges.
}
\end{frame}

\begin{frame}{Exercise 1: one-counter systems}
\Margines{
\tiny
A \emph{one-counter VASS} consists of a finite set $Q$ of control states
and transitions
\[
  p\xrightarrow{a,d}q,\qquad d\in\mathbb Z.
\]
Its configurations are $(q,n)\in Q\times\mathbb N$, and
\[
  (p,n)\xrightarrow{a}(q,n+d)
\]
is enabled iff $n+d\geq 0$.

\begin{enumerate}
\item Consider the one-state system
\[
q\xrightarrow{\mathsf{inc},+1}q,
\qquad
q\xrightarrow{\mathsf{dec},-1}q.
\]
Draw the reachable part of the induced LTS from $(q,0)$ up to counter
value $4$.

\item Give the same system the $\mathbb Z$-semantics, i.e.\ let the counter
range over $\mathbb Z$ and remove the non-negativity test. Draw the part
between counter values $-3$ and $3$.

\item Compare the two trace sets. Which traces are removed by the
$\mathbb N$-semantics?
\end{enumerate}

\begin{tikzpicture}[->,node distance=22mm,semithick]
  \node[state,initial] (q) {$q$};
  \path
    (q) edge[loop above] node {$\mathsf{inc},+1$} (q)
    (q) edge[loop below] node {$\mathsf{dec},-1$} (q);
\end{tikzpicture}
\end{center}
}
\end{frame}

\begin{frame}{Exercise 3: a pushdown automaton}
\Margines{
\tiny
\begin{definition}
A \emph{labelled pushdown automaton} (PDA) is a tuple
\[
  \mathcal P=(Q,\Sigma,\Gamma,\Delta,q_0,\bot),
\]
where $Q$ is a finite set of control states, $\Sigma$ is a finite alphabet,
$\Gamma$ is a finite stack alphabet, $\bot\in\Gamma$ is the bottom symbol,
and
\[
  \Delta\subseteq Q\times\Gamma\times\Sigma\times Q\times\Gamma^{\leq 2}.
\]
A configuration is $(q,w)\in Q\times\Gamma^*$, with the top of the stack
at the right end. A transition $(p,A,a,q,\beta)\in\Delta$ induces
\[
  (p,wA)\xrightarrow{a}(q,w\beta).
\]
Thus $\beta=\varepsilon$ pops $A$, $|\beta|=1$ replaces it, and
$|\beta|=2$ can push one additional symbol. We never pop $\bot$.
\end{definition}

Let $Q=\{q\}$ and $\Gamma=\{0,1,\bot\}$. From every stack content allow
$\mathsf{push}_0$ and $\mathsf{push}_1$; allow $\mathsf{pop}$ whenever
the top symbol is not $\bot$.

\begin{enumerate}
\item Starting from $(q,\bot)$, draw the induced LTS up to stack height $3$.
\item Ignore edge directions and identify the underlying graph.
\item Add a second control state $r$ and transitions that change
$q\leftrightarrow r$ without changing the stack. Redraw the first two levels.
\end{enumerate}

\begin{center}
\begin{tikzpicture}[->,semithick,node distance=14mm and 18mm]
\node[state] (e) {$\bot$};
\node[state,below left=of e] (z) {$\bot0$};
\node[state,below right=of e] (o) {$\bot1$};
\node[state,below left=of z] (zz) {$\bot00$};
\node[state,below right=of z] (zo) {$\bot01$};
\node[state,below left=of o] (oz) {$\bot10$};
\node[state,below right=of o] (oo) {$\bot11$};

\path
(e) edge[bend right=8] node[left] {\scriptsize $0$} (z)
(z) edge[bend right=8] node[right] {\scriptsize pop} (e)
(e) edge[bend left=8] node[right] {\scriptsize $1$} (o)
(o) edge[bend left=8] node[left] {\scriptsize pop} (e)
(z) edge (zz)
(z) edge (zo)
(o) edge (oz)
(o) edge (oo);
\end{tikzpicture}
\end{center}
}
\end{frame}

\begin{frame}{Exercise 4: treewidth of a PDA configuration graph}
\Margines{
\tiny
\begin{definition}
A \emph{tree decomposition} of an undirected graph $G=(V,E)$ is a tree
$T$ whose nodes are labelled by bags $B_t\subseteq V$ such that:
\begin{enumerate}
\item every vertex of $G$ belongs to some bag;
\item for every edge $\{u,v\}\in E$, some bag contains both $u$ and $v$;
\item for every $v\in V$, the bags containing $v$ form a connected
subtree of $T$.
\end{enumerate}
Its width is $\max_t |B_t|-1$, and $\operatorname{tw}(G)$ is the minimum
possible width.
\end{definition}

\medskip
Assume the pushdown system is in the normal form from Exercise~3.

\begin{enumerate}
\item Use the tree $\Gamma^*$ of stack words as the decomposition tree.
\item For every $w\in\Gamma^*$ define
\[
B_w=
   \bigl(Q\times\{w\}\bigr)
   \cup
   \bigcup_{A\in\Gamma} \bigl(Q\times\{wA\}\bigr).
\]
Prove that $(B_w)_{w\in\Gamma^*}$ is a tree decomposition of the
underlying configuration graph.
\item Deduce the bound
\[
 \operatorname{tw}(G)\leq |Q|(|\Gamma|+1)-1.
\]
\end{enumerate}

\begin{block}{Courcelle's theorem --- version sufficient here}
For every fixed $k$ and every fixed MSO sentence $\varphi$, model checking
$\varphi$ on finite graphs of treewidth at most $k$ is linear in the size
of the graph (given a suitable tree decomposition).
\end{block}
}
\end{frame}

\begin{frame}{Exercise 7: a workflow net}
\Margines{
\tiny
A \emph{workflow net} has a unique input place $i$ with no incoming arcs,
a unique output place $o$ with no outgoing arcs, and every node lies on a
directed path from $i$ to $o$.

\begin{center}
\begin{tikzpicture}[node distance=9mm and 14mm,semithick]
\tikzset{
  pplace/.style={circle,draw,minimum size=6mm,inner sep=0pt},
  ptrans/.style={rectangle,draw,fill=black,minimum width=2mm,minimum height=7mm,inner sep=0pt}
}
\node[pplace] (i) {$i$};
\node[ptrans,right=of i] (s) {};
\node[pplace,above right=of s] (p1) {$p_1$};
\node[pplace,below right=of s] (p2) {$p_2$};
\node[ptrans,right=of p1] (t1) {};
\node[ptrans,right=of p2] (t2) {};
\node[pplace,right=of t1] (q1) {$q_1$};
\node[pplace,right=of t2] (q2) {$q_2$};
\node[ptrans,below right=of q1] (j) {};
\node[pplace,right=of j] (o) {$o$};
\draw[->] (i)--node[above]{$s$}(s);
\draw[->] (s)--(p1);
\draw[->] (s)--(p2);
\draw[->] (p1)--node[above]{$t_1$}(t1)--(q1);
\draw[->] (p2)--node[below]{$t_2$}(t2)--(q2);
\draw[->] (q1)--(j);
\draw[->] (q2)--(j);
\draw[->] (j)--node[above]{$j$}(o);
\fill ($(i.center)+(-1.2pt,0)$) circle (1.1pt);
\end{tikzpicture}
\end{center}

The initial marking contains one token in $i$.

\begin{enumerate}
\item Verify that the net is a workflow net.
\item Draw its complete reachability LTS.
\item Show that from every reachable marking it is possible to reach the
      final marking with one token in $o$ and no token elsewhere.
\item Can a token remain anywhere else when $o$ is marked?  Is every transition
      executable in some run?  Conclude that this workflow net is sound.
\item Add a ``short-circuit'' transition $o\to i$. Is the resulting net a
      T-system?
\end{enumerate}
}
\end{frame}

\begin{frame}{Büchi automata: definition and warm-up}
\Margines{
\tiny
\begin{definition}
A nondeterministic Büchi automaton is
\[
  A=(Q,\Sigma,\delta,Q_0,F),
\]
where $\delta\subseteq Q\times\Sigma\times Q$.
A run
\[
q_0\xrightarrow{a_0}q_1\xrightarrow{a_1}q_2\xrightarrow{a_2}\cdots
\]
is accepting if it visits $F$ infinitely often.
The language $L(A)\subseteq\Sigma^\omega$ consists of words having
an accepting run.
\end{definition}

\begin{exampleblock}{Warm-up}
Let $\Sigma=\{a,b\}$. Prove that the automaton below accepts exactly the
words containing infinitely many occurrences of $a$.
\end{exampleblock}

\begin{center}
\begin{tikzpicture}[->,node distance=32mm,semithick]
\node[state,initial] (q0) {$q_0$};
\node[state,accepting,right=of q0] (q1) {$q_1$};
\path
(q0) edge[loop above] node {$b$} (q0)
(q0) edge[bend left] node {$a$} (q1)
(q1) edge[loop above] node {$a$} (q1)
(q1) edge[bend left] node {$b$} (q0);
\end{tikzpicture}
\end{center}
}
\end{frame}

\begin{frame}{Exercise 8: construct Büchi automata I}
\Margines{
\tiny
Let $\Sigma=\{a,b\}$. Construct a Büchi automaton for each language.

\begin{enumerate}
\item Words containing infinitely many occurrences of $a$.
\item Words containing only finitely many occurrences of $a$.
\item Words containing infinitely many $b$ such that, between every two
\emph{consecutive} $b$'s, the number of $a$'s is divisible by $3$.
\item Words such that, between every two consecutive $b$'s, the number
of $a$'s is divisible by $3$. There is no requirement that $b$ occurs
infinitely often.
\end{enumerate}

\medskip
The following finite-state ``modulo $3$'' component is useful in
parts 3--4.

\begin{center}
\begin{tikzpicture}[->,node distance=24mm,semithick]
\node[state,initial] (r0) {$0$};
\node[state,right=of r0] (r1) {$1$};
\node[state,right=of r1] (r2) {$2$};
\path
(r0) edge[bend left=15] node {$a$} (r1)
(r1) edge[bend left=15] node {$a$} (r2)
(r2) edge[bend left=35] node[below] {$a$} (r0)
(r0) edge[loop above] node {$b$} (r0);
\end{tikzpicture}
\end{center}

For every construction, state explicitly which states are accepting and
explain why the Büchi condition enforces the required infinitary property.
}
\end{frame}

\begin{frame}{Exercise 9: construct Büchi automata II}
\Margines{
\tiny
Again let $\Sigma=\{a,b\}$.

\begin{enumerate}
\setcounter{enumi}{4}
\item The number of $b$'s is infinite, and only finitely many gaps
between consecutive $b$'s contain a number of $a$'s divisible by $3$.

Equivalently: from some point on, every gap between consecutive $b$'s
has length non-zero modulo $3$.

\item The number of $b$'s is finite, and every gap between consecutive
$b$'s contains a number of $a$'s divisible by $3$.

\item The number of $b$'s is finite, and for every two \emph{distinct}
occurrences of $b$, the number of $a$'s between them is not divisible
by $3$.
\end{enumerate}

\begin{block}{Hint for part 7}
Associate with each occurrence of $b$ the number of $a$'s seen so far,
modulo $3$. What does the condition say about these residues?
In particular, how many occurrences of $b$ can there be?
\end{block}

\begin{block}{General design principle}
Separate a finite-state \emph{safety condition} (forbidden local patterns)
from the Büchi/co-Büchi style requirement saying that something happens
infinitely often or only finitely often.
\end{block}
}
\end{frame}

\begin{frame}{Exercise 10: proving that a language is not Büchi-recognisable}
\Margines{
\tiny
Call $L\subseteq\Sigma^\omega$ \emph{$\omega$-regular} if it is recognised
by a Büchi automaton.

Prove the following two lemmas.

\begin{lemma}[ultimately periodic witness]
If $L\neq\emptyset$ is $\omega$-regular, then there exist
$u\in\Sigma^*$ and $v\in\Sigma^+$ such that $uv^\omega\in L$.
\end{lemma}

\begin{lemma}[fixed-tail section]
If $L$ is $\omega$-regular and $x\in\Sigma^\omega$ is fixed, then
\[
 K_x=\{u\in\Sigma^*:ux\in L\}
\]
is a regular language of finite words.
\end{lemma}

\medskip
Use the lemmas to prove that the following languages are not
$\omega$-regular:
\begin{enumerate}
\item
\[
L_1=\left\{a\,b\,a^2b\,a^3b\,a^4b\cdots\right\}.
\]
\item
\[
L_2=
\left\{
a^{n_1}ba^{n_2}ba^{n_3}b\cdots:
n_1<n_2<n_3<\cdots
\right\}.
\]
\item Over $\{a,b,c\}$,
\[
L_3=\{a^n b^n c^\omega:n\geq 1\}.
\]
For part 3 use the section with tail $c^\omega$ and intersect it with
$a^+b^+$.
\end{enumerate}
}
\end{frame}

\begin{frame}{Exercise 11: Kripke structures as Büchi automata}
\Margines{
\tiny
\begin{definition}
A Kripke structure is
\[
K=(S,s_0,R,AP,\ell),
\]
where $S$ is a finite set of states, $R\subseteq S\times S$ is total,
$AP$ is a finite set of atomic propositions, and
$\ell:S\to 2^{AP}$ labels states.

Its traces are
\[
\operatorname{Tr}(K)=
\left\{
\ell(s_0)\ell(s_1)\ell(s_2)\cdots:
s_0Rs_1Rs_2R\cdots
\right\}
\subseteq (2^{AP})^\omega.
\]
\end{definition}

\begin{enumerate}
\item Construct a Büchi automaton $A_K$ such that
\[
L(A_K)=\operatorname{Tr}(K).
\]
\item Your construction may use one fresh initial state and make all
states corresponding to $S$ accepting.
\item Prove both inclusions.
\end{enumerate}

\begin{center}
\begin{tikzpicture}[->,node distance=18mm,semithick,scale=0.72,transform shape]
\node[state,initial] (s0) {$s_0:\{p\}$};
\node[state,right=of s0] (s1) {$s_1:\emptyset$};
\node[state,below=of s1] (s2) {$s_2:\{p\}$};
\path
(s0) edge[loop above] (s0)
(s0) edge (s1)
(s1) edge[bend left] (s2)
(s2) edge[bend left] (s1)
(s2) edge[bend left=35] (s0);
\end{tikzpicture}
\end{center}

Apply your construction explicitly to the example above.
}
\end{frame}

\begin{frame}{Exercise 12: intersection with traces --- finite case}
\Margines{
\tiny
Let
\[
A=(Q_A,\Sigma,\delta,Q_A^0,F)
\]
be a Büchi automaton and
\[
B=(C,\Sigma,\longrightarrow,c_0)
\]
an LTS.

Define their synchronous product. Its configurations are pairs $(q,c)$ and
\[
(q,c)\xrightarrow{a}(q',c')
\]
iff
\[
q\xrightarrow{a}_A q'
\qquad\text{and}\qquad
c\xrightarrow{a}_B c'.
\]
A product configuration is Büchi-accepting iff $q\in F$.

\begin{enumerate}
\item Prove:
\[
L(A)\cap\operatorname{Tr}(B)\neq\emptyset
\]
iff the product has an infinite run visiting $F\times C$ infinitely often.
\item Assume $B$ is finite. Show that non-emptiness is equivalent to
reachability of a strongly connected component that contains an accepting
product state and contains a directed cycle.
\item Give a graph algorithm for the problem.
\end{enumerate}

\begin{block}{Convention}
If $B$ is presented as an NFA, use its transition graph as the LTS.
The final states of the NFA are irrelevant unless an additional condition
explicitly refers to them.
\end{block}
}
\end{frame}

\begin{frame}{Exercise 13: intersection with traces --- infinite-state systems}
\Margines{
\tiny
Let $A$ be a finite Büchi automaton. For each representation of $B$ below,
form the synchronous product of $A$ with the induced LTS of $B$:
\begin{enumerate}
\item one-counter VASS;
\item one-counter automaton;
\item $d$-dimensional VASS;
\item pushdown automaton.
\end{enumerate}

For each model:

\begin{enumerate}
\item[(a)] Describe a configuration of the product.
\item[(b)] Show that the product remains in the same model class:
only the finite control is multiplied by $Q_A$.
\item[(c)] Reformulate
\[
L(A)\cap\operatorname{Tr}(B)=\emptyset
\]
as a repeated-reachability/Büchi problem in the product.
\item[(d)] Identify which part of the original infinite-state semantics
(counter non-negativity, zero tests, stack operations) is unchanged by
the product.
\end{enumerate}

\begin{block}{Main observation}
A finite specification automaton does not add a new unbounded data
structure. It only refines the finite control of the system.
\end{block}
}
\end{frame}
