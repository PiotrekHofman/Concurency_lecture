\section{CCS as an algebra}
% 4
\begin{frame}{A fragment of CCS}
  \begin{block}{Syntax}
  \[
    P ::= \nil \mid \alpha.P \mid P+P \mid P\parallel P
       \mid \blockact{H}{P},\qquad \alpha\in\Sigma\cup\{\tau\}.
  \]
  \end{block}
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{@{}p{.25\textwidth}p{.67\textwidth}@{}}
    $\alpha.P$ & Prefix: first $\alpha$, then $P$.\\
    $P+Q$ & Alternative initial transitions (nondeterministic choice).\\
    $P\parallel Q$ & Interleaving plus synchronisation of $a$ with $\bar a$.\\
    $\blockact{H}{P}$ & Block labels in $H$; $\tau$ is never blocked.
  \end{tabular}
  \Source{Same notation as Lecture 5; standard CCS restriction of a channel corresponds to blocking both $a$ and $\bar a$.}
  \note{This is recursion-free CCS with explicit action-blocking notation retained from L5. No fairness assumption.}
\end{frame}

% Recursive process definitions
\begin{frame}{Recursive process definitions}
  \begin{block}{Defining a process constant}
    \[
      X \;=\; a.X+b.\nil
    \]
    This equation defines $X$: it can do $a$ and return to $X$, or do $b$ and stop.
  \end{block}
  \begin{exampleblock}{Reading definitions as transitions}
    Unfold the right-hand side to find the transitions:
    \[
      X\step{a}X
      \qquad
      X\step{b}\nil.
    \]
    More generally, $X=P$ gives $X$ the transitions of $P$, with occurrences of $X$ retained as process states.
  \end{exampleblock}
  \Takeaway{Here $=$ introduces a recursive definition; it does not mean bisimilarity.}
  \note{The recursion is guarded: every recursive occurrence lies beneath an action prefix. We extend the recursion-free syntax with process constants and guarded defining equations.}
\end{frame}

% Basic SOS rules
\begin{frame}{SOS rules: prefix, choice, and blocking}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.55\textwidth}
      \begin{block}{Prefix and choice}
        \[
          \frac{}{\alpha.P\step{\alpha}P}
          \qquad
          \frac{P\step{\alpha}P'}{P+Q\step{\alpha}P'}
          \qquad
          \frac{Q\step{\alpha}Q'}{P+Q\step{\alpha}Q'}
        \]
      \end{block}
      \begin{exampleblock}{No transition from $\nil$}
        The process $\nil$ has no outgoing transitions.
      \end{exampleblock}
    \end{column}
    \begin{column}{.42\textwidth}
      \begin{block}{Blocking}
        For $H\subseteq\Sigma$:
        \[
          \frac{P\step{\alpha}P'\qquad \alpha\notin H}
          {\blockact{H}{P}\step{\alpha}\blockact{H}{P'}}
        \]
      \end{block}
      \begin{exampleblock}{Effect of $H$}
        A visible action in $H$ is suppressed; $\tau$ is never blocked.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \Takeaway{Choice inherits either branch's transitions; blocking removes selected visible transitions without relabelling the rest.}
  \note{The two choice rules are symmetric. Since $H$ contains only visible labels, the blocking rule also passes $\tau$ transitions.}
\end{frame}

% Finite-state transition systems as processes
\begin{frame}{Encoding a finite automaton as a process}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.43\textwidth}
      \centering
      \begin{tikzpicture}[node distance=16mm and 18mm]
        \node[st] (q0) {$q_0$};
        \node[st,right=of q0] (q1) {$q_1$};
        \node[terminal,below=of q0] (end) {$\nil$};
        \draw[trans] (q0) to[bend left=12] node[lab,above] {$a$} (q1);
        \draw[trans] (q1) to[bend left=12] node[lab,below] {$b$} (q0);
        \draw[trans] (q1) -- node[lab,right] {$c$} (end);
      \end{tikzpicture}
    \end{column}
    \begin{column}{.55\textwidth}
      \begin{block}{One equation per state}
        \[
          Q_0=a.Q_1
          \qquad
          Q_1=b.Q_0+c.\nil
        \]
        The start state is $Q_0$. Each automaton edge becomes a prefixed summand; no outgoing edges means $\nil$.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \Takeaway{The process LTS reproduces the automaton's labelled transitions.}
  \Source{This encoding preserves transition structure; accepting/final-state status is not represented by this CCS fragment.}
  \note{Stress the correspondence state-by-state: a state equation lists exactly its outgoing edges. To preserve language acceptance, the process language needs an explicit success observation in addition to these transitions.}
\end{frame}

% 5
\begin{frame}{One rule, three kinds of transitions}
  \begin{block}{Interleaving}
    \[
      \frac{P\step{\alpha}P'}{P\parallel Q\step{\alpha}P'\parallel Q}
      \qquad
      \frac{Q\step{\beta}Q'}{P\parallel Q\step{\beta}P\parallel Q'}
    \]
  \end{block}
  \begin{exampleblock}{Communication}
    \[
      \frac{P\step{a}P'\quad Q\step{\bar a}Q'}
      {P\parallel Q\step{\tau}P'\parallel Q'}
    \]
  \end{exampleblock}
  \Takeaway{A handshake is \emph{possible}, not compulsory, until exposed actions are blocked.}
  \note{The three derivations are exactly the new parallel SOS rules of Lecture 5. Emphasise that prefix, choice and blocking were already introduced.}
\end{frame}

\begin{frame}{Blocking forces an internal handshake}
  \[
     S=\blockact{H}{a.b.\nil\parallel\bar a.c.\nil},
     \qquad H=\{a,\bar a\}.
  \]
  \vspace{2mm}
  \centering
  \begin{tikzpicture}[x=1cm,y=1cm]
    \node[st,minimum size=9mm] (s) at (0,0) {$S$};
    \node[st,minimum size=9mm] (t) at (4,0) {$T$};
    \draw[tauedge] (s)--node[lab,above] {$\tau$}(t);
    \node[font=\small,text=Muted,below=5mm of t,align=center] {$T=\blockact{H}{b.\nil\parallel c.\nil}$};
  \end{tikzpicture}
  \vspace{3mm}
  \begin{block}{Why only one transition?}
    $a$ and $\bar a$ are blocked individually; their synchronisation produces $\tau$, which survives.
  \end{block}
  \note{Assume all visible names a,b,c are distinct and b,c do not form a complementary pair. This avoids unnecessary handshake complications in T.}
\end{frame}


\begin{frame}{A VASS-style counter from parallel processes}
  \begin{block}{Counter components}
    For each counter $i\in\{x,y\}$, use a generator and indistinguishable tokens:
    \[
      G_i = u_i.(G_i\parallel T_i)
      \qquad
      T_i = d_i.\nil.
    \]
    The generator creates one token on $u_i$; a token disappears on $d_i$.
  \end{block}
  \begin{exampleblock}{Control states request one update}
    A control-state prefix $\bar u_i$ synchronises with $G_i$ to increment
    counter $i$; $\bar d_i$ synchronises with one $T_i$ to decrement it.
    The decrement is unavailable when there are no tokens.
  \end{exampleblock}
  \Takeaway{Parallel composition represents the counter value as the number of token processes.}
  \note{These are finite recursive equations, but unfolding G_i can create unboundedly many parallel T_i components. This is how the example avoids being finite-state.}
\end{frame}

\begin{frame}{Two VASS transitions, each changing one counter}
  \begin{block}{Finite control}
    \[
      Q_0=\bar u_x.Q_1
      \qquad
      Q_1=\bar d_y.Q_0
    \]
    From $Q_0$, increment $x$; from $Q_1$, decrement $y$ and return to $Q_0$.
  \end{block}
  \begin{exampleblock}{Compose and restrict}
    Let $H$ contain each $u_i,\bar u_i,d_i,\bar d_i$ for $i=x,y$.
    The system is
    \[
      \blockact{H}{Q_0\parallel G_x\parallel G_y
        \parallel T_x^{\,m}\parallel T_y^{\,n}}.
    \]
    Only the requested handshake can occur; it becomes $\tau$. In particular,
    the decrement step is disabled when $n=0$.
  \end{exampleblock}
  \Takeaway{This illustrates VASS-style counter updates, not a full machine encoding.}
  \Source{Here $\blockact{H}{\cdot}$ is CCS restriction (blocking visible actions in $H$); some presentations call this hiding.}
  \note{Each transition changes exactly one counter and the finite control state. T_i^k denotes k parallel copies, with T_i^0=\nil.}
\end{frame}

\begin{frame}{When may we write that $P,$ $Q$ are the same in CCS calculations?}
  \begin{block}{Equations}
\begin{itemize}
  \item Set of equations. 
  \item We transform $P$ into $Q$ using equations.
  \item Equations preserve (bisimulation, weak bisimulation).
\end{itemize}    
  \end{block}
  Example: 
  $P+P \equiv P$

  We can prove that: 
  $P+P+P \equiv P$
\visible<2->{
\begin{block}{Why equations?}
\begin{itemize}
  \item Algebraic proofs are often simpler than bisimulation proofs.
  \item Equations are parametric in $P$, so they can be applied to any process.
\end{itemize}
\end{block}}
\end{frame}

% 9
\begin{frame}{Four basic equations for choice}
  \begin{align*}
    P+Q &\equiv Q+P &&\text{(commutativity)}\\[2mm]
    (P+Q)+R &\equiv P+(Q+R) &&\text{(associativity)}\\[2mm]
    P+P &\equiv P &&\text{(idempotence)}\\[2mm]
    P+\nil &\equiv P &&\text{(zero)}
  \end{align*}
  \vspace{3mm}
  \Source{These choice laws are part of the equational axiomatisation for the recursion-free CCS fragment used in this lecture.}
  \note{One can justify idempotence by relating P+P to P and their derivatives. No law a.(P+Q)=a.P+a.Q is included!}
\end{frame}

% 10
\begin{frame}{Useful laws for parallelism and blocking}
  \begin{align*}
    P\parallel\nil &\equiv P,\\[3mm]
    \blockact{H}{\nil}&\equiv\nil,\\[3mm]
    \blockact{H}{P+Q}&\equiv \blockact{H}{P}+\blockact{H}{Q}.
  \end{align*}
  \begin{block}{Blocking a prefix}
    \[
      \blockact{H}{\alpha.P}\equiv
      \begin{cases}
        \alpha.\blockact{H}{P},&\alpha\notin H,\\
        \nil,&\alpha\in H.
      \end{cases}
    \]
  \end{block}
  \Takeaway{All equations preserve \emph{strong} bisimilarity.}
  \note{H is a set of visible labels, not containing tau; so tau always takes the first branch. The P||0 law follows from a simple coinductive relation.}
\end{frame}

% 11
\begin{frame}{A tempting \emph{false} law}
  \[
  \blockact{H}{P\parallel Q}
  \quad\not\sim\quad
  \blockact{H}{P}\parallel\blockact{H}{Q}.
  \]
  \vspace{2mm}
  \begin{block}{Counterexample}
    \[
      P=a.\nil,\quad Q=\bar a.\nil,\quad H=\{a,\bar a\}.
    \]
    \[
      \underbrace{\blockact{H}{a.\nil\parallel\bar a.\nil}}_{\step{\tau}\text{ enabled}}
      \qquad\qquad
      \underbrace{\blockact{H}{a.\nil}\parallel\blockact{H}{\bar a.\nil}}_{\text{deadlocked}}.
    \]
  \end{block}
  \Takeaway{Blocking \emph{after} composing preserves the possibility of synchronisation.}
  \note{This is the central anti-law. Students often expect restriction to distribute over every constructor.}
\end{frame}

% 12
\begin{frame}{The CCS expansion theorem}
  \vspace{1mm}
  Suppose $P=\sum_{i\in I}\alpha_i.P_i$ and $Q=\sum_{j\in J}\beta_j.Q_j$.
  \begin{block}{Expand the initial moves of $P\parallel Q$}
    \vspace{-2mm}
    \begin{align*}
    P\parallel Q\ \equiv\ {}&
      \sum_{i\in I}\alpha_i.(P_i\parallel Q)\\[-.5mm]
      &+\sum_{j\in J}\beta_j.(P\parallel Q_j)\\[-.5mm]
      &+\sum_{\substack{i\in I,\,j\in J\\\alpha_i=\bar\beta_j}}
           \tau.(P_i\parallel Q_j).
    \end{align*}
  \end{block}
  \centering
  \small \textcolor{Teal}{left moves}\qquad +\qquad
  \textcolor{Indigo}{right moves}\qquad +\qquad
  \textcolor{Warn}{synchronisations}
  \Source{CCS expansion theorem; the equation denotes strong bisimilarity under standard interleaving SOS.}
  \note{For finite sums, the formula follows by comparing the initial-transition sets. Sum over empty set means 0. Do not extend a fairness interpretation without qualification.}
\end{frame}

\begin{frame}{Expansion without communication: first step}
  Let $a$ and $b$ be distinct, non-complementary actions.
  \[
     P=a.\nil\parallel b.\nil.
  \]
  \begin{block}{Apply the expansion theorem}
    \[
      P\equiv
        a.(\nil\parallel b.\nil)
        +b.(a.\nil\parallel\nil).
    \]
  \end{block}
  \centering
  \begin{tikzpicture}[x=1cm,y=1cm]
    \node[st] (p) at (0,1.2) {$P$};
    \node[st] (l) at (-2.2,-.35) {$\nil\parallel b.\nil$};
    \node[st] (r) at (2.2,-.35) {$a.\nil\parallel\nil$};
    \draw[trans] (p)--node[lab,left] {$a$}(l);
    \draw[trans] (p)--node[lab,right] {$b$}(r);
  \end{tikzpicture}
  \note{There is no communication term: neither a nor b is the co-action of the other.}
\end{frame}

% 14
\begin{frame}{Expansion without communication: normal form}
  \begin{align*}
     a.\nil\parallel b.\nil
       &\equiv a.(\nil\parallel b.\nil)
             +b.(a.\nil\parallel\nil)\\[2mm]
       &\equiv \boxed{a.b.\nil+b.a.\nil}.
  \end{align*}
  \vspace{2mm}
  \begin{exampleblock}{Interpretation}
    Two independent single actions admit precisely two sequential schedules, $ab$ and $ba$.
  \end{exampleblock}
  \Takeaway{For finite recursion-free CCS, expansion eliminates parallel composition in favour of prefix and choice.}
  \note{Distinguish expressing the same interleaving LTS from claiming causality/independence is preserved by the syntax: later true-concurrency semantics can make this distinction observable.}
\end{frame}

\begin{frame}{Adding guarded recursion to the proof system}
  \begin{block}{RDP: unfold a recursive definition}
    If $P$ is defined by $P\coloneqq F(P)$, then
    \[
      P\equiv F(P).
    \]
  \end{block}
  \begin{exampleblock}{RSP: use uniqueness of guarded solutions}
    \[
      \frac{P:= F(P)\qquad Q:= F(Q)}
           {P\sim Q}
      \qquad\text{if every recursive occurrence in $F$ is action-guarded.}
    \]
  \end{exampleblock}
  \Takeaway{RDP unfolds; RSP proves two solutions of the same guarded specification equivalent.}
  \note{These principles extend equational reasoning with guarded recursive specifications. The relation symbol in the premises and conclusion is strong bisimilarity here. RSP is an inference rule justified by uniqueness, not an ordinary unconditional equation.}
\end{frame}

\begin{frame}{Example: uniqueness proves a recursive equivalence}
  Suppose the recursive definitions are
  \[
    P\coloneqq a.P
    \qquad\qquad
    Q\coloneqq a.Q.
  \]
  \begin{block}{Apply RDP and RSP}
    \[
      P:= a.P,\qquad Q:= a.Q
      \quad\Longrightarrow_{\mathrm{RSP}}\quad
      P\sim Q.
    \]
  \end{block}
  \begin{exampleblock}{Why the guard condition matters}
    For strong bisimilarity, a $\tau$-prefix can guard recursion. For weak
    bisimilarity, $\tau$ alone is not enough:
    \[
      \nil\wb\tau.\nil,\qquad
      a.\nil\wb\tau.a.\nil,\qquad
      \nil\not\wb a.\nil.
    \]
    Thus, $X=\tau.X$ has unequivalent weak solutions; weak uniqueness results
    require stronger guarding conditions.
  \end{exampleblock}
  \note{RSP applies to the shared guarded equation $X=a.X$, giving the result immediately. For the weak example, both $\nil$ and $a.\nil$ satisfy $X\wb\tau.X$, but they are not weakly bisimilar to one another; visible guarding is therefore needed for the corresponding weak uniqueness principles.}
\end{frame}

\begin{frame}{Rooted weak bisimilarity: the root condition}
  \small
  \begin{block}{Matching an initial transition}
    For visible $a$ and the silent action $\tau$, write
    \[
      \xRightarrow{a}_{\!r} \;=\;
        \xrightarrow{\tau *}\xrightarrow{a}\xrightarrow{\tau *},
      \qquad
      \xRightarrow{\tau}_{\!r} \;=\;
        \xrightarrow{\tau +}.
    \]
    In particular, an initial $\tau$ must be matched by
    \emph{at least one} $\tau$-step.
  \end{block}
  \begin{block}{Definition (observation congruence)}
    $P\rwb Q$ iff, for each $\alpha\in\Sigma\cup\{\tau\}$,
    \[
       P\step{\alpha}P'
       \ \Longrightarrow\
       \exists Q':\ Q\xRightarrow{\alpha}_{\!r}Q'
                     \ \land\ P'\wb Q',
    \]
    and symmetrically for every initial move of $Q$.
  \end{block}
  \vspace{-1mm}
  \Takeaway{$\tau.a.\nil\not\rwb a.\nil$,
    but $\tau.\tau.a.\nil\rwb\tau.a.\nil$.
    The successors need only be \emph{weakly} bisimilar.}
  \note{Replace the old slide: it required a matching single step and
    used R on successors, effectively stating strong bisimulation.
    Explain that the root is special. For an observable a, allow
    tau-star before and after a. For initial tau, demand tau-plus;
    after matching, ordinary weak bisimilarity suffices.
    Source: van Glabbeek, A Characterisation of Weak Bisimulation
    Congruence, Section 2.}
\end{frame}



\begin{frame}{Hennessy--Milner laws for silent steps}
  \begin{block}{Three laws for rooted weak bisimilarity}
    \small
    \[
      \begin{aligned}
        \text{(A5)}\quad
          &X+\tau.X \;\equiv\; \tau.X,\\[1mm]
        \text{(A6.1)}\quad
          &\mu.(X+\tau.Y)
            \;\equiv\; \mu.(X+\tau.Y)+\mu.Y,\\[1mm]
        \text{(A6.2)}\quad
          &\mu.\tau.X \;\equiv\; \mu.X,
          \qquad \mu\in\Sigma\cup\{\tau\}.
      \end{aligned}
    \]
  \end{block}
  \Takeaway{These laws abstract internal steps without identifying $\tau.P$ with $P$ at the root.}
  \Source{M. Hennessy and R. Milner, \emph{Algebraic Laws for Nondeterminism and Concurrency} (1985).}
  \note{Present these as the three tau-laws A5, A6.1, and A6.2. The action parameter mu ranges over visible actions and tau. They are sound equations for rooted weak (observational) bisimilarity; do not replace them with the unrestricted law tau.P=P.}
\end{frame}




\begin{frame}{Recursion: uniqueness of solutions}
  \footnotesize

  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \begin{block}{Strong bisimilarity}
        Guarded recursion is sufficient.

        Every recursive variable occurs under
        an action prefix $\mu$, including $\tau$.
      \end{block}
    \end{column}

    \begin{column}{.48\textwidth}
      \begin{block}{Rooted weak bisimilarity}
        Classical sufficient conditions:
        \begin{itemize}
          \item \textbf{Strongly guarded:}
            recursive variables occur under
            visible action prefixes ($a\neq\tau$).
          \item \textbf{Sequential:}
            no parallel composition or restriction
            on paths to recursive variables.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \begin{block}{Recursive Specification Principle (RSP)}
    If $F$ is strongly guarded and sequential, then
    \[
      \boxed{
        \frac{
          P := F(P)
          \qquad
          Q := F(Q)
        }{
          P \approx_{\mathrm{rw}} Q
        }
      }
    \]
    Example: $P := a.P$ and
    $Q := a.Q$
    imply $P \approx_{\mathrm{rw}} Q$.
  \end{block}



  \note{
    Contrast with strong bisimilarity, where
    guarding by tau is sufficient.
    For rooted weak bisimilarity, consider X = tau.X.
    Both P = tau.0 and Q = tau.a.0 satisfy this equation
    up to rooted weak bisimilarity, but P and Q
    are not equivalent.
    Strong guarding and sequentiality are
    sufficient conditions, not necessary ones.
  }
\end{frame}

\begin{frame}{Why both restrictions on recursion are needed}
  \footnotesize
  Let $P=\tau.0$ and $Q=\tau.b.0$, where
  $b\notin\{a,\bar a\}$. Then $P\not\approx_{\mathrm{rw}}Q$.

  \begin{block}{1. Guarded, but not strongly guarded}
    \[
      X=\tau.X
    \]
    Both $P$ and $Q$ are solutions, since
    $R\approx_{\mathrm{rw}}\tau.R$ for $R\in\{P,Q\}$
    (using $\tau.\tau.S\approx_{\mathrm{rw}}\tau.S$).
  \end{block}

  \begin{block}{2. Strongly guarded, but not sequential}
    \[
      X=F(X),\qquad
      F(X)=(a.X\mid\bar a.0)\setminus\{a\}
    \]
    Synchronization turns the visible guard into $\tau$;
    restriction blocks independent $a$ and $\bar a$.
    For $R\in\{P,Q\}$:
    \[
       F(R)\sim\tau.R\approx_{\mathrm{rw}}R.
    \]
  \end{block}

  \begin{alertblock}{Takeaway}
    Neither ordinary guarding nor strong guarding alone
    guarantees uniqueness up to rooted weak bisimilarity.
  \end{alertblock}
\end{frame}

\begin{frame}{Theorem: bisimulation is a congruence here}
  \begin{block}{Congruence theorem}
    For $\star\in\{\mathrm{s},\mathrm{w},\mathrm{rw}\}$,
    \[
      P\equiv_{\star}Q
      \quad\Longrightarrow\quad
      C[P]\equiv_{\star}C[Q]
      \qquad \text{for every legal }C[-].
    \]
  \end{block}
  \vspace{-1mm}
  \textbf{Proof for weak bisimilarity.} Let $P\equiv_{\mathrm w}Q$ and fix $\sigma$.
  \begin{enumerate}\setlength\itemsep{2.7mm}
    \item \textbf{Assignment prefix.} Put $\sigma'=\sigma[x\mapsto n]$.
      \[
        \Store{x:=n;P}{\sigma}\step{\tau}\Store{P}{\sigma'},
        \qquad
        \Store{x:=n;Q}{\sigma}\step{\tau}\Store{Q}{\sigma'}.
      \]
    \item \textbf{Output prefix.} Both sides emit $\operatorname{out}(\sigma(x))$
      and continue in $\Store{P}{\sigma}$ and $\Store{Q}{\sigma}$.
  \end{enumerate}
  \vspace{-1mm}
  \Takeaway{The first moves match, and their targets are equivalent. Induct on $C[-]$.}
  \note{For the assignment case, P equiv_w Q implies weak bisimilarity of continuations at sigma', because equivalence was quantified over every store. For output, first observable labels coincide and the successors are weakly bisimilar at sigma. The root step of either prefix is exactly the same label on each side, so the same proof also covers rooted weak bisimilarity (weak-equivalent derivatives), as well as strong bisimilarity (strong-equivalent derivatives). Induction on the grammar of C[-] gives the desired congruence.}
\end{frame}

\begin{frame}{The invariant: the last assignment wins}
  Let $\sigma$ be an arbitrary initial store. Define
  \[
     \sigma_1=\sigma[x\mapsto 1],\qquad
     \sigma_2=\sigma[x\mapsto 2].
  \]
  \begin{block}{The key identity}
    \[
       \underbrace{\sigma[x\mapsto 1][x\mapsto 2]}_{\text{two writes}}
       =
       \underbrace{\sigma[x\mapsto 2]}_{\text{one write}}.
    \]
  \end{block}
  \Takeaway{The intermediate stores differ, but the stores at the next observation agree.}
  \note{This is what the simulation relation must remember: the x=1 state is acceptable because its next transition is a write to x, not a read.}
\end{frame}

\begin{frame}{When the optimisation becomes incorrect}
  Suppose each write is observable (e.g. a volatile shared location).
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.46\textwidth}
      \begin{alertblock}{Before}
        \[
        \operatorname{wr}(x,1)\,\operatorname{wr}(x,2)\,\operatorname{out}(2)
        \]
      \end{alertblock}
    \end{column}
    \begin{column}{.46\textwidth}
      \begin{exampleblock}{After}
        \[
         \operatorname{wr}(x,2)\,\operatorname{out}(2)
        \]
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{5mm}
  \[
    \operatorname{Tr}(\text{before})\ne\operatorname{Tr}(\text{after}).
  \]
  \Takeaway{No weak bisimulation can justify this replacement with visible write labels.}
  \Source{In CompCert, a volatile store to a global variable generates an observable event.}
  \note{An observing concurrent thread is another issue (intermediate values could be exposed); the toy language did not include concurrency. Volatile accesses provide a direct label-level counterexample.}
\end{frame}

% 36
\begin{frame}{What makes dead-store elimination sound?}
  \begin{enumerate}\setlength\itemsep{5mm}
    \item The removed write is \textbf{not observable}.
    \item Its right-hand side has \textbf{no relevant side effect or fault}.
    \item The old value is \textbf{not read before it is overwritten}.
    \item The transformation respects \textbf{aliasing and the memory model}.
  \end{enumerate}
  \vspace{1mm}
  \Takeaway{Real optimisers prove a \emph{general invariant}; our proof established one instance.}
  \note{This is a local toy example, not a complete correctness proof for all DSE programs. In real intermediate representations liveness/neededness dataflow analysis and memory safety are essential.}
\end{frame}

\begin{frame}{Why not require bisimulation for every pass?}

\begin{frame}[fragile]{Optimization: shrink the critical section}
  \small

void record(const Job& j) {
  std::lock_guard<std::mutex> g(mu);
  auto k = slow_hash(j.id) % N;
  ++counts[k];
}
\end{verbatim}

      \end{alertblock}

    \begin{column}{.49\textwidth}
      \begin{exampleblock}{After: compute before taking the mutex}
        \vspace{1mm}

\begin{verbatim}
void record(const Job& j) {
  auto k = slow_hash(j.id) % N;
  std::lock_guard<std::mutex> g(mu);

\end{verbatim}

      \end{exampleblock}
    \end{column}
  \end{columns}

  \begin{block}{Cost of the protected region (schematic)}
    \[
      T_{\rm held}^{\rm before}\approx t_{\rm hash}+t_{\rm update},
      \qquad
      T_{\rm held}^{\rm after}\approx t_{\rm update}.
    \]
  \end{block}
  \Takeaway{Move \emph{independent local work} out of the critical section;
    keep the shared update under the mutex.}
  \note{Job j has a stable, immutable id; slow_hash is deterministic,
    terminating, pure and non-throwing. The counts array is shared and
    is accessed under the mutex. This is a realistic lock-contention
    optimization, not an optimization of total computational work.}
\end{frame}

\begin{frame}{CCS model and equational correctness}
  \small
  Let $k=h(j)$. Model request reception by a visible action
  $\mathrm{req}(j)$; $\ell,u$ denote lock/unlock handshakes,
  and $\tau$ denotes the local computation of $k$.
  \begin{block}{The two implementations}
    \vspace{-2mm}
    \[
      \begin{aligned}
        P &= \mathrm{req}(j).\ell.\tau.\mathrm{inc}(k).u.\nil,\\[1mm]
        Q &= \mathrm{req}(j).\tau.\ell.\mathrm{inc}(k).u.\nil.
      \end{aligned}
    \]
    \vspace{-2mm}
  \end{block}
  \begin{exampleblock}{Derivation using A6.2: $\mu.\tau.X\rwb\mu.X$}
    \vspace{-2mm}
    \[
      \begin{aligned}
        P &\rwb \mathrm{req}(j).\ell.\mathrm{inc}(k).u.\nil
          &&(\mu=\ell),\\[1mm]
          &\rwb Q
          &&(\mu=\mathrm{req}(j)).
      \end{aligned}
    \]
    \vspace{-2mm}
  \end{exampleblock}
  {\footnotesize\color{Warn}
    The common visible request is essential:
    $\ell.\tau.R\not\rwb\tau.\ell.R$ in general.\par}
  \vspace{2mm}
  \Takeaway{$\boxed{P\rwb Q}$: the protected update is preserved,
    while a silent calculation moves across lock acquisition.}
  \note{The action req(j) represents the observable method invocation;
    it is common to both processes. Without this common initial visible
    prefix, ell.tau.R and tau.ell.R are weakly bisimilar but need not be
    rooted weakly bisimilar. A6.2 eliminates the tau following ell
    on the left and the tau following req(j) on the right. We have
    abstracted a total pure function, not a computation that can throw
    or diverge.}
\end{frame}

\begin{frame}{Correctness in a concurrent system}
  \small
  \begin{block}{An abstract mutex and arbitrary other workers}
    \[
      M=\bar\ell.\bar u.M,\qquad
      H=\{\ell,\bar\ell,u,\bar u\},
    \]
    \[
      \mathcal C[-]
      =\blockact{H}{[-]\parallel M\parallel W}.
    \]
    $M$ permits one acquisition at a time; restriction blocks
    unmatched lock/unlock actions.
  \end{block}
  \begin{exampleblock}{Use congruence instead of checking every schedule}
    \[
      P\rwb Q
      \quad\Longrightarrow\quad
      \boxed{\mathcal C[P]\rwb\mathcal C[Q]}.
    \]
    Rooted weak bisimilarity is preserved by CCS parallel composition
    and restriction.
  \end{exampleblock}
  \Takeaway{The equation proves correctness in \emph{all CCS contexts}
    of this form; lower lock-hold time is a separate cost claim.}
  \note{W denotes other concurrent workers obeying the mutex protocol.
    The protocol M is a simplified token mutex, not a formal account of
    the C++ memory model, fairness, timed locking, or exceptions.
    The congruence step is valid for arbitrary W in the modeled CCS
    language. The model assumes each protected increment is an atomic
    event inc(k) and that all requests use stable data.}
\end{frame}

\begin{frame}[fragile]{When is critical-section shrinking unsafe?}
  \small
  \begin{alertblock}{Unsafe transformation: move a shared read outside the mutex}
    \vspace{1mm}
    \scriptsize
\begin{verbatim}
// UNSAFE transformation:
auto k = slow_hash(shared_key);  // read outside mu
std::lock_guard<std::mutex> g(mu);
++counts[k % N];
\end{verbatim}
    \vspace{-1mm}
  \end{alertblock}

  \begin{block}{What goes wrong?}
    Another thread may update \texttt{shared\_key} while
    \texttt{slow\_hash} reads it. Without additional synchronization,
    this can introduce a \textbf{data race} (undefined behaviour in C++).
  \end{block}

  \Takeaway{The moved computation must be independent of mutable
    shared state. A local computation is not necessarily a safe $\tau$-step.}

  \note{The example breaks the assumption that the hash only reads
    immutable or thread-local data. The original read was protected,
    whereas the moved read can race with another writer.}
\end{frame}

\begin{frame}{When is critical-section shrinking correct?}
  \small
  \begin{block}{Assumptions required by the CCS proof}
    \begin{itemize}\setlength\itemsep{3mm}
      \item Input data are \textbf{immutable or thread-local}.
      \item The hash is \textbf{pure, deterministic, terminating,
            and non-throwing}.
      \item The shared update remains protected by the
            \textbf{same mutex}.
      \item Timing, lock timeouts, and scheduler fairness are
            \textbf{not observable} in this CCS model.
    \end{itemize}
  \end{block}
  \begin{exampleblock}{What has been proved?}
    Rooted weak bisimilarity establishes preservation of the
    modelled observable behaviour. It does \emph{not} establish
    a reduction in running time.
  \end{exampleblock}
  \Takeaway{Correctness is relative to the observation model;
    performance improvement is a separate claim.}

  \note{Pure means no shared-memory effects, I/O, or synchronization.
    An operation that can throw or diverge cannot be replaced by the
    local terminating tau-step used in our proof. The lock-hold time
    decreases in the cost model; system speedup depends on contention.}
\end{frame}
