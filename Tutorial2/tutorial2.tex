\section{Automata and Expressions for Infinite Words}

\paragraph{Büchi Automata.}
A \emph{nondeterministic automaton} is a tuple
\[
\mathcal A=(Q,\Sigma,\delta,q_0,\mathscr{F}),
\]
where $Q$ is a finite set of states, $\Sigma$ is a finite alphabet,
$\delta\subseteq Q\times\Sigma\times Q$ is the transition relation,
$q_0\in Q$ is the initial state, and $\mathscr{F}$ is some sort of \emph{acceptance condition}.
For a run
\[
\rho : q_0\xrightarrow{a_0}q_1\xrightarrow{a_1}q_2
\xrightarrow{a_2}\cdots
\]
we shall write
\[
\operatorname{Inf}(\rho)
=
\{q\in Q:q=q_i\text{ for infinitely many }i\}.
\]


In a \emph{Büchi automaton} (NBA), 
the acceptance condition is generated by a set $F \subseteq Q$ of states,
and 
a run $\rho$
is \emph{accepting} if it visits $F$ infinitely often:
$\operatorname{Inf}(\rho)\cap F \neq \varnothing$.
%We write $\L_B(\mathcal A)$ for the language accepted by $\mathcal A$.



\paragraph{Other automata.}
A \emph{Muller automaton} is equipped with an
acceptance family
\(
\mathcal F\subseteq 2^Q
\).
A run $\rho$ is accepting if
\(
\operatorname{Inf}(\rho)\in\mathcal F.
\)

Similarly, 
a \emph{Rabin automaton} is an automaton equipped with an
acceptance condition
\[
\mathcal R=\{(E_1,F_1),\ldots,(E_k,F_k)\},
\qquad E_i,F_i\subseteq Q.
\]
A run $\rho$ is accepting if there exists $i\in\{1,\ldots,k\}$
such that
\[
\operatorname{Inf}(\rho)\cap E_i=\varnothing
\quad
\text{and}
\quad
\operatorname{Inf}(\rho)\cap F_i\ne\varnothing,
\]
i.e. if it visits an ``allowed'' set $E_i$ infinitely often,
and avoids a ``forbidded'' set co-finitely often 
(co-finite: always, excpet a finite number of times).

Finally, 
a \emph{parity automaton} is an automaton equipped with a priority
function
\[
\Omega:Q\longrightarrow\{0,\ldots,d\},
\]
and a run $\rho$ is accepting if
\[
\min\{\Omega(q):q\in\operatorname{Inf}(\rho)\}
\]
is even.


\paragraph{$\omega$-Regular Languages}
An \emph{$\omega$-regular expression} over an alphabet $\Sigma$ is
defined inductively by the grammar
\[
s ::= r^\omega \mid rs \mid s+s,
\]
where $r$ ranges over \emph{regular expressions} over $\Sigma$.
Its language $\L(s)\subseteq\Sigma^\omega$ is defined inductively by
\[
\begin{aligned}
\L(r^\omega) &= \L(r)^\omega,\\
\L(rs)       &= \L(r)\L(s),\\
\L(s_1+s_2)  &= \L(s_1)\cup\L(s_2).
\end{aligned}
\]
A language $L\subseteq\Sigma^\omega$ is called \emph{$\omega$-regular}
if $L=\L(s)$ for some $\omega$-regular expression $s$.

\section{Exercises}


\begin{exercise}[Constructing Büchi automata]
Let $\Sigma=\{a,b\}$. Construct Büchi automata recognizing the
following languages.
\begin{enumerate}
\item Words containing infinitely many occurrences of $a$.
\item Words containing only finitely many occurrences of $a$.
\item Words containing infinitely many occurrences of both $a$ and $b$.
\item Words in which every occurrence of $a$ is eventually followed
by an occurrence of $b$.
\item Words containing infinitely many occurrences of the factor $ab$.
\item Words that are ultimately periodic with period $ab$:
\[
\{u(ab)^\omega:u\in\Sigma^*\}.
\]
\item Words containing infinitely many $a$'s, but no two consecutive
occurrences of $a$.
\item Words having some suffix belonging to $(a^*b)^\omega$.
\end{enumerate}
Which of these languages can be recognized by deterministic Büchi
automata?
\end{exercise}


\begin{exercise}[Pumping lemma for Büchi automata]
Let $L\subseteq\Sigma^\omega$ be recognized by a Büchi automaton
with $n$ states. For every $w\in L$, there exist
$x,y\in\Sigma^*$ and $z\in\Sigma^\omega$ such that
\[
w=xyz,\qquad |xy|\le n,\qquad |y|>0,
\]
and
\[
xy^kz\in L
\qquad\text{for every }k\ge0.
\]
\end{exercise}

\begin{proof}
Let $w=a_0a_1a_2\cdots\in L$, and consider an accepting run
\[
q_0\xrightarrow{a_0}q_1
\xrightarrow{a_1}q_2
\xrightarrow{a_2}\cdots.
\]
Among the first $n+1$ states $q_0,\ldots,q_n$, two must coincide.
Thus there exist $0\le i<j\le n$ such that $q_i=q_j$.

Write
\[
x=a_0\cdots a_{i-1},\qquad
y=a_i\cdots a_{j-1},\qquad
z=a_ja_{j+1}\cdots.
\]
Since $q_i=q_j$, the portion of the run reading $y$ is a cycle.
Consequently, we may traverse this cycle any finite number of
times, obtaining a run on $xy^kz$ for every $k\ge0$.

The resulting run is accepting, since it differs from the original
accepting run only by a finite prefix. Hence $xy^kz\in L$ for
every $k\ge0$.
\end{proof}

\begin{exercise}[Non-$\omega$-regular languages]
Use the pumping lemma to prove that the following languages are
not $\omega$-regular.
\begin{enumerate}
\item
\[
L_1=\{a^nb^nc^\omega:n\ge1\}.
\]
\item
\[
L_2=\{a^nb^{2n}(ab)^\omega:n\ge1\}.
\]
\item
\[
L_3=\{a^nb^mc^\omega:n,m\ge1,\ n<m<2n\}.
\]
\item
\[
L_4=
\{a^{n_1}ba^{n_2}ba^{n_3}b\cdots:
n_i=n_1\text{ for every }i\ge1\}.
\]
\end{enumerate}
\end{exercise}

\begin{exercise}
Every nonempty language recognized by a Büchi automaton contains
an ultimately periodic word, i.e., a word of the form
\[
uv^\omega,\qquad
u,v\in\Sigma^*,\quad v\ne\varepsilon.
\]
\end{exercise}

\begin{proof}
Consider an accepting run. Some accepting state occurs infinitely
often. Choose two occurrences of that state and repeat the finite
segment of the run between them forever.
\end{proof}

\begin{exercise}[Languages without ultimately periodic words]
Prove that the following languages are not $\omega$-regular.
\begin{enumerate}
\item The language of infinite words over $\{a,b\}$ containing
infinitely many $b$'s and arbitrarily long consecutive blocks of $a$'s.
\item
\[
L=\left\{
a^{n_1}ba^{n_2}ba^{n_3}b\cdots:
n_1<n_2<n_3<\cdots
\right\}.
\]
\end{enumerate}
\end{exercise}

\subsection{Other Acceptance Conditions}

\begin{exercise}[Acceptance conditions]
Let $Q=\{p,q,r\}$. Describe which runs are accepted by each of
the following conditions.
\begin{enumerate}
\item The Muller condition
\[
\mathcal F=\{\{p\},\{p,q\},\{q,r\}\}.
\]
\item The Rabin condition
\[
\mathcal R=\{(\{p\},\{q\}),(\{q\},\{r\})\}.
\]
\item The parity condition
\[
\Omega(p)=0,\qquad
\Omega(q)=1,\qquad
\Omega(r)=2.
\]
\item Give a Muller condition equivalent to the Rabin condition
in (b).
\item Give a Rabin condition equivalent to the parity condition
in (c).
\end{enumerate}
\end{exercise}

\subsection{Equivalence of Acceptance Conditions}

\begin{exercise}[Büchi acceptance as other acceptance conditions]
Let $\mathcal A$ be a Büchi automaton with accepting set $F$.
Show that its acceptance condition can be expressed as follows.
\begin{enumerate}
\item As a Muller condition:
\[
\mathcal F=\{S\subseteq Q:S\cap F\ne\varnothing\}.
\]
\item As a Rabin condition consisting of the single pair
\[
(\varnothing,F).
\]
\item As a parity condition:
\[
\Omega(q)=
\begin{cases}
0,&q\in F,\\
1,&q\notin F.
\end{cases}
\]
\end{enumerate}
\end{exercise}

\begin{exercise}[Parity to Rabin]
Let $\mathcal A$ be a parity automaton with priority function
\[
\Omega:Q\to\{0,\ldots,d\}.
\]
For every even priority $2i\le d$, define
\[
E_i=\{q:\Omega(q)<2i\},
\qquad
F_i=\{q:\Omega(q)=2i\}.
\]
Prove that the Rabin condition consisting of the pairs
$(E_i,F_i)$ is equivalent to the original parity condition.
\end{exercise}

\begin{exercise}[Rabin to Muller]
Let
\[
\mathcal R=\{(E_i,F_i):1\le i\le k\}
\]
be a Rabin acceptance condition. Define
\[
\mathcal F=
\left\{
S\subseteq Q:
\exists i\
(S\cap E_i=\varnothing
\ \land\
 S\cap F_i\ne\varnothing)
\right\}.
\]
Prove that the Muller condition $\mathcal F$ is equivalent to
$\mathcal R$.
\end{exercise}

\begin{exercise}[Rabin to Büchi]
Show that every language recognized by a nondeterministic Rabin
automaton is recognized by a nondeterministic Büchi automaton.

\emph{Hint.}
Nondeterministically choose a Rabin pair $(E_i,F_i)$ and guess
a position after which no state from $E_i$ is visited. From that
position onward, require infinitely many visits to $F_i$.
\end{exercise}

\begin{exercise}[Muller to Büchi]
Show that every language recognized by a nondeterministic Muller
automaton is recognized by a nondeterministic Büchi automaton.

\emph{Hint.}
Nondeterministically select $S\in\mathcal F$, guess a position
after which the run visits only states of $S$, and ensure that
every state of $S$ occurs infinitely often.
\end{exercise}

\begin{exercise}
The following classes of languages coincide:
\[
\mathrm{NBA}
=
\mathrm{NPA}
=
\mathrm{NRA}
=
\mathrm{NMA}.
\]
\end{exercise}

\begin{proof}
The preceding exercises establish
\[
\mathrm{NBA}\subseteq\mathrm{NPA}
\subseteq\mathrm{NRA}
\subseteq\mathrm{NMA}
\subseteq\mathrm{NBA}.
\]
\end{proof}

\remark
The analogous equivalence does not hold for deterministic Büchi
automata. For example,
\[
L=\{w\in\{a,b\}^\omega:
\text{$a$ occurs only finitely often}\}
\]
is recognized by a deterministic parity automaton, but not by
any deterministic Büchi automaton.


\paragraph{Büchi's characterization}
For every $L\subseteq\Sigma^\omega$, the following are equivalent:
\begin{enumerate}
\item $L$ is $\omega$-regular.
\item $L$ is recognized by a nondeterministic Büchi automaton.
\end{enumerate}

\begin{exercise}[Büchi's characterization]
Prove the preceding theorem.

\begin{enumerate}
\item
Let $U,V\subseteq\Sigma^*$ be regular languages, with
$\varepsilon\notin V$. Construct a Büchi automaton recognizing
\[
UV^\omega.
\]

\emph{Hint.}
Combine finite automata for $U$ and $V$. The Büchi automaton
should read a prefix in $U$, then repeatedly read nonempty
words in $V$, visiting an accepting state after each repetition.

Explain how to extend the construction to finite unions.

\item
Let
\[
\mathcal A=(Q,\Sigma,\delta,q_0,F)
\]
be a Büchi automaton. For $p,q\in Q$, define
\[
L_{p,q}
=
\{u\in\Sigma^*:
\text{there exists a path from $p$ to $q$ labeled $u$}\}.
\]
Observe that each $L_{p,q}$ is regular.

Prove that
\[
\L_\omega(\mathcal A)
=
\bigcup_{q\in F}
L_{q_0,q}
\bigl(L_{q,q}\setminus\{\varepsilon\}\bigr)^\omega.
\]

\emph{Hint.}
In an accepting run, some accepting state occurs infinitely
often. Split the run at successive occurrences of that state.
\end{enumerate}
\end{exercise}

\subsection{Decision Problems}

The fundamental decision problems for Büchi automata are:
\begin{enumerate}
\item \emph{Emptiness:}
given $\mathcal A$, decide whether
\[
\L_\omega(\mathcal A)=\varnothing.
\]
\item \emph{Intersection emptiness:}
given $\mathcal A,\mathcal B$, decide whether
\[
\L_\omega(\mathcal A)\cap\L_\omega(\mathcal B)
=\varnothing.
\]
\item \emph{Universality:}
given $\mathcal A$, decide whether
\[
\L_\omega(\mathcal A)=\Sigma^\omega.
\]
\end{enumerate}

\begin{exercise}[Emptiness]
The emptiness problem for Büchi automata is decidable.

More precisely, a Büchi automaton accepts some word if and only
if its transition graph contains a reachable directed cycle
passing through an accepting state.

What is the complexity of deciding Emptiness of Büchi automata?
[Hint: linear time in the size of the transition graph.]
\end{exercise}

\begin{proof}
Suppose $\mathcal A$ accepts some word, and let $\rho$ be an
accepting run. Some accepting state $q\in F$ appears infinitely
often in $\rho$. Hence $q$ is reachable from the initial state,
and there exists a nonempty path from $q$ back to $q$.

Conversely, suppose there is a path from $q_0$ to an accepting
state $q$, together with a nonempty path from $q$ back to $q$.
Let $u$ and $v$ label these paths, respectively. Then
\[
uv^\omega
\]
is accepted by $\mathcal A$.
\end{proof}

\begin{exercise}[Emptiness algorithm]
Design an algorithm deciding whether a given Büchi automaton
has a nonempty language.
\begin{enumerate}
\item Describe an algorithm based on strongly connected components.
\item Prove its correctness.
\item Analyze its time complexity.
\item If the language is nonempty, show how to construct an
accepted word of the form $uv^\omega$.
\end{enumerate}
\end{exercise}

\begin{exercise}[Intersection]
Let $\mathcal A_1,\mathcal A_2$ be Büchi automata.

Prove that there exists a Büchi automaton $\mathcal B$ such that
\[
\L_\omega(\mathcal B)
=
\L_\omega(\mathcal A_1)
\cap
\L_\omega(\mathcal A_2).
\]

\emph{Hint.}
The naive synchronous product with accepting set
$F_1\times F_2$ does not work. Add a component that alternates
between waiting for a visit to $F_1$ and waiting for a visit
to $F_2$.

\begin{enumerate}
\item Generalize the construction to the intersection of $k$
Büchi languages.
\item Deduce that intersection emptiness is decidable.
\item Determine the size of your construction.
\end{enumerate}
\end{exercise}

\begin{exercise}[Universality]
Consider the universality problem for nondeterministic Büchi
automata:
\[
\L_\omega(\mathcal A)=\Sigma^\omega\;?
\]
\begin{enumerate}
\item Explain why the usual subset construction for finite
automata does not immediately solve the problem.
\item Assuming that Büchi automata are effectively closed under
complementation, prove that universality is decidable.
\item Show that universality is decidable for complete
deterministic Büchi automata without constructing a Büchi
automaton for the complement.

\emph{Hint.}
A deterministic Büchi automaton rejects a word if its unique
run visits the accepting states only finitely often.
\end{enumerate}
\end{exercise}

\subsection{Closure Under Complementation}

\begin{exercise}[Complementing Büchi automata]
For finite words, a complete deterministic finite automaton can
be complemented by replacing its accepting set $F$ with
$Q\setminus F$.

Show that this construction does not generally work for
deterministic Büchi automata.

\emph{Hint.}
Consider a run visiting both $F$ and $Q\setminus F$ infinitely
often.
\end{exercise}

\begin{exercise}[Complementing deterministic Muller automata]
Let
\[
\mathcal A=(Q,\Sigma,\delta,q_0,\mathcal F)
\]
be a complete deterministic Muller automaton.

Define
\[
\overline{\mathcal F}=2^Q\setminus\mathcal F.
\]

Prove that the automaton
\[
\overline{\mathcal A}
=
(Q,\Sigma,\delta,q_0,\overline{\mathcal F})
\]
recognizes the complement of $\L_\omega(\mathcal A)$.
\end{exercise}

\begin{exercise}[Complementing deterministic parity automata]
Let $\mathcal A$ be a complete deterministic parity automaton
with priority function
\[
\Omega:Q\to\mathbb N.
\]
Define
\[
\Omega'(q)=\Omega(q)+1.
\]
Prove that replacing $\Omega$ with $\Omega'$ gives an automaton
recognizing
\[
\Sigma^\omega\setminus\L_\omega(\mathcal A).
\]
\end{exercise}


\paragraph{McNaughton's determinization theorem}
Every nondeterministic Büchi automaton has an equivalent
deterministic Muller automaton.

\paragraph{Closure under complementation}
If $L\subseteq\Sigma^\omega$ is $\omega$-regular, then
\[
\Sigma^\omega\setminus L
\]
is $\omega$-regular.

\begin{proof}
By Büchi's characterization, there exists a nondeterministic
Büchi automaton $\mathcal A$ recognizing $L$.

By McNaughton's theorem, there exists an equivalent deterministic
Muller automaton $\mathcal B$.

Complementing the Muller acceptance family of $\mathcal B$
produces a deterministic Muller automaton recognizing
$\Sigma^\omega\setminus L$.

Finally, every language recognized by a Muller automaton is
recognized by a nondeterministic Büchi automaton. Hence
$\Sigma^\omega\setminus L$ is $\omega$-regular.
\end{proof}

\subsection{Further Exercises}

\begin{exercise}[Boolean closure]
Prove that the class of $\omega$-regular languages is closed under
\[
\cup,\qquad\cap,\qquad\setminus,
\]
and symmetric difference.
\end{exercise}

\begin{exercise}[Finite prefixes]
Let $L\subseteq\Sigma^\omega$ be $\omega$-regular and
$K\subseteq\Sigma^*$ be regular.

Prove that
\[
KL=\{uv:u\in K,\ v\in L\}
\]
is $\omega$-regular.
\end{exercise}

\begin{exercise}[Prefix independence]
Call an $\omega$-language $L$ \emph{prefix-independent} if
\[
uw\in L\iff w\in L
\]
for every $u\in\Sigma^*$ and $w\in\Sigma^\omega$.

\begin{enumerate}
\item Prove that the following languages are prefix-independent:
\begin{enumerate}
\item words containing infinitely many $a$'s;
\item words containing finitely many $a$'s;
\item words containing infinitely many occurrences of $ab$.
\end{enumerate}
\item Give an example of an $\omega$-regular language that is
not prefix-independent.
\item Show that a language recognized by a complete deterministic
Muller, Rabin, or parity automaton need not be prefix-independent.
\end{enumerate}
\end{exercise}

\begin{exercise}[Ultimately periodic words]
Let
\[
\mathrm{UP}
=
\{uv^\omega:u,v\in\Sigma^*,\ v\ne\varepsilon\}.
\]

Prove that two $\omega$-regular languages
$L_1,L_2\subseteq\Sigma^\omega$ are equal if and only if
\[
L_1\cap\mathrm{UP}
=
L_2\cap\mathrm{UP}.
\]

\emph{Hint.}
If $L_1\ne L_2$, their symmetric difference is a nonempty
$\omega$-regular language. Apply the proposition asserting
the existence of ultimately periodic words.
\end{exercise}
