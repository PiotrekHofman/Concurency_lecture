\section{Automata and Expressions for Infinite Words}

\paragraph{Büchi Automata.}
A \emph{nondeterministic automaton} is a tuple
\[
\mathcal A=(Q,\Sigma,\delta,q_0,\mathscr{F}),
\]
where $Q$ is a finite set of states, $\Sigma$ is a finite alphabet,
$\delta\subseteq Q\times\Sigma\times Q$ is the transition relation,
$q_0\in Q$ is the initial state, and $\mathscr{F}$ is some sort of \emph{acceptance condition}.
For a run
\[
\rho : q_0\xrightarrow{a_0}q_1\xrightarrow{a_1}q_2
\xrightarrow{a_2}\cdots
\]
we shall write
\[
\operatorname{Inf}(\rho)
=
\{q\in Q:q=q_i\text{ for infinitely many }i\}.
\]

\noindent
In a \emph{Büchi automaton} (NBA), 
the acceptance condition is generated by a set $F \subseteq Q$ of states,
and 
a run $\rho$
is \emph{accepting} if it visits $F$ infinitely often:
$\operatorname{Inf}(\rho)\cap F \neq \varnothing$.
%We write $\L_B(\mathcal A)$ for the language accepted by $\mathcal A$.



\paragraph{Other automata.}
There are other automata that are similar to Büchi,
with different acceptance conditions.
\begin{itemize}[leftmargin=*]
\item A \emph{Muller automaton} is equipped with an
acceptance family
\(
\mathcal F\subseteq 2^Q
\).
A run $\rho$ is accepting if
the set of infinitely occurring states is an element of the family $\mathcal F$:
\(
\operatorname{Inf}(\rho)\in\mathcal F.
\)

\item A \emph{Rabin automaton} is an automaton equipped with an
acceptance condition
\[
\mathcal R=\{(E_1,F_1),\ldots,(E_k,F_k)\},
\qquad E_i,F_i\subseteq Q.
\]
A run $\rho$ is accepting if there exists $i\in\{1,\ldots,k\}$
such that
\[
\operatorname{Inf}(\rho)\cap E_i=\varnothing
\quad
\text{and}
\quad
\operatorname{Inf}(\rho)\cap F_i\ne\varnothing,
\]
i.e. if it visits an ``allowed'' set $E_i$ infinitely often,
and avoids a ``forbidded'' set co-finitely often 
(co-finitely: always, excpet a finite number of times).

\item A \emph{parity automaton} is an automaton equipped with a priority
function
\(
\Omega:Q\longrightarrow\{0,\ldots,d\},
\)
and a run $\rho$ is accepting if
\[
\min\{\Omega(q):q\in\operatorname{Inf}(\rho)\}
\]
is even.
\end{itemize}

\paragraph{$\omega$-Regular Languages.}
An \emph{$\omega$-regular expression} over an alphabet $\Sigma$ is
defined inductively by the grammar
\[
s ::= r^\omega \mid rs \mid s+s,
\]
where $r$ ranges over \emph{regular expressions} over $\Sigma$.
Its language $\L(s)\subseteq\Sigma^\omega$ is defined inductively by
\[
\begin{aligned}
\L(r^\omega) &= \L(r)^\omega,\\
\L(rs)       &= \L(r)\L(s),\\
\L(s_1+s_2)  &= \L(s_1)\cup\L(s_2).
\end{aligned}
\]
A language $L\subseteq\Sigma^\omega$ is called \emph{$\omega$-regular}
if $L=\L(s)$ for some $\omega$-regular expression $s$.

Equivalently, a language is $\omega$-regular if, and only if, it can be recognized
by a nondeterministic Büchi automaton.



\paragraph{McNaughton's determinization theorem.}
It is a difficult theorem that every nondeterministic Büchi automaton 
is \emph{equivalent to} (i.e., it recognizes the same language as)
a \emph{deterministic} Muller automaton.

Among other results, this allows us to prove the closure of
$\omega$-regular languages under complementation:
if $L\subseteq\Sigma^\omega$ is $\omega$-regular, then
\(
\Sigma^\omega\setminus L
\)
is also $\omega$-regular.

\begin{proof}
By Büchi's characterization, there exists a nondeterministic
Büchi automaton $\mathcal A$ recognizing $L$.
By McNaughton's theorem, there exists an equivalent deterministic
Muller automaton $\mathcal B$.
Complementing the Muller acceptance family of $\mathcal B$
produces a deterministic Muller automaton recognizing
$\Sigma^\omega\setminus L$.
Finally, every language recognized by a Muller automaton is
recognized by a nondeterministic Büchi automaton. Hence
$\Sigma^\omega\setminus L$ is $\omega$-regular.
\end{proof}

\paragraph{Usual Automata Problems.}
The fundamental decision problems for Büchi automata\footnote{\ldots and all kinds of automata, in general!}
are:
\begin{enumerate}
\item \emph{Emptiness:}
given $\mathcal A$, decide whether
\(\L_\omega(\mathcal A)=\varnothing\).
\item \emph{Disjointness:}
given $\mathcal A,\mathcal B$, decide whether
\(\L_\omega(\mathcal A)\cap\L_\omega(\mathcal B)
=\varnothing\).
\item \emph{Universality:}
given $\mathcal A$, decide whether
\(\L_\omega(\mathcal A)=\Sigma^\omega\).
\end{enumerate}

\begin{exercise}[Constructing Büchi automata]
Let $\Sigma=\{a,b\}$. Construct Büchi automata recognizing the
following languages.
\begin{enumerate}
\item Words containing infinitely many occurrences of $a$.
\item Words containing only finitely many occurrences of $a$.
\item Words containing infinitely many occurrences of both $a$ and $b$.
\item Words in which every occurrence of $a$ is eventually followed
by an occurrence of $b$.
\item Words containing infinitely many $a$'s, but no two consecutive
occurrences of $a$.
\item Words containing infinitely many $b$ such that, between every two
\emph{consecutive} $b$'s, the number of $a$'s is divisible by $3$.
\item Words such that, between every two consecutive $b$'s, the number
of $a$'s is divisible by $3$. There is no requirement that $b$ occurs
infinitely often.

\item The number of $b$'s is infinite, and only finitely many gaps
between consecutive $b$'s contain a number of $a$'s divisible by $3$.
[\textit{Hint}: equivalently, from some point on, every gap between consecutive $b$'s
has length non-zero modulo $3$.]

\item The number of $b$'s is finite, and every gap between consecutive
$b$'s contains a number of $a$'s divisible by $3$.

\item The number of $b$'s is finite, and for every two \emph{distinct}
occurrences of $b$, the number of $a$'s between them is not divisible
by $3$. [\textit{Hint}: how many $b$'s can there be?]
\end{enumerate}
Which of these languages can be recognized by deterministic Büchi
automata?
\end{exercise}

\begin{exercise}[Parity to Rabin]
Let $\mathcal A$ be a parity automaton with priority function
{\({\Omega:Q\to\{0,\ldots,d\}}\).}
For every even priority $2i\le d$, define
\[
E_i=\{q:\Omega(q)<2i\},
\qquad
F_i=\{q:\Omega(q)=2i\}.
\]
Prove that the Rabin condition consisting of the pairs
$(E_i,F_i)$ is equivalent to the original parity condition of $\mathcal A$.
\end{exercise}

\begin{exercise}[Rabin to Muller]
Let \(\mathcal R=\{(E_i,F_i):1\le i\le k\}\)
be a Rabin acceptance condition. Define
\[
\mathcal F=
\left\{
S\subseteq Q \ : \ \exists i\ (S\cap E_i=\varnothing \ \land\ S\cap F_i\neq\varnothing)
\right\}.
\]
Prove that the Muller condition $\mathcal F$ is equivalent to
$\mathcal R$.
\end{exercise}

\begin{exercise}[Büchi's characterization]
Prove that for every $L\subseteq\Sigma^\omega$, 
the following are equivalent:
\begin{enumerate}
\item $L$ is $\omega$-regular.
\item $L$ is recognized by a nondeterministic Büchi automaton.
\end{enumerate}

\noindent[\textit{Hint for (2)$\Rightarrow$(1)}:
Let $U,V\subseteq\Sigma^*$ be regular languages, with
$\varepsilon\notin V$. Construct a Büchi automaton recognizing
\(UV^\omega\)
by combining the finite automata for $U$ and $V$. The Büchi automaton
should read a prefix in $U$, then repeatedly read nonempty
words in $V$, visiting an accepting state after each repetition.
Explain how to extend the construction to finite unions.]
\bigskip

\noindent[\textit{Hint for (1)$\Rightarrow$(2)}:
Let \(\mathcal A=(Q,\Sigma,\delta,q_0,F)\)
be a Büchi automaton. For $p,q\in Q$, define
\[
L_{p,q}
=
\{u\in\Sigma^*:
\text{there exists a path from $p$ to $q$ labeled $u$}\}.
\]
Observe that each $L_{p,q}$ is regular.
Prove that
\[
\L_\omega(\mathcal A)=
\bigcup_{q\in F}L_{q_0,q}\left(L_{q,q}\setminus\{\varepsilon\}\right)^\omega.]
\]

\end{exercise}

\begin{exercise}[Prefix independence]
Call an $\omega$-language $L$ \emph{prefix-independent} if
\[
uw\in L\iff w\in L
\]
for every $u\in\Sigma^*$ and $w\in\Sigma^\omega$.

\begin{enumerate}
\item Prove that the following languages are prefix-independent:
\begin{enumerate}
\item words containing infinitely many $a$'s;
\item words containing finitely many $a$'s;
\item words containing infinitely many occurrences of $ab$.
\end{enumerate}
\item Give an example of an $\omega$-regular language 
that is not prefix independent.
\end{enumerate}
\end{exercise}

\begin{exercise}[Myhill--Nerode equivalences for $\omega$-languages]
Let $L\subseteq\Sigma^\omega$. Determine which of the following
equivalence relations $\sim_L$ have the property that
\[
L\text{ is $\omega$-regular}
\quad\Longleftrightarrow\quad
\sim_L\text{ has finite index}.
\]

\begin{enumerate}
\item The equivalence relation on $\Sigma^*$ defined by
\[
u\sim_L v
\iff
\forall w\in\Sigma^\omega:
\bigl(uw\in L\iff vw\in L\bigr).
\]

\item The equivalence relation on $\Sigma^\omega$ defined by
\[
u\sim_L v
\iff
\forall w\in\Sigma^*:
\bigl(wu\in L\iff wv\in L\bigr).
\]

\item The equivalence relation on $\Sigma^*$ defined by
\[
u\sim_L v
\iff
\begin{cases}
uw\in L\iff vw\in L
&\text{for all }w\in\Sigma^\omega,\\
s(ut)^\omega\in L\iff s(vt)^\omega\in L
&\text{for all }s,t\in\Sigma^*
\end{cases}
\]
where the second condition is understood only when
$ut,vt\ne\varepsilon$.
\end{enumerate}

\noindent[\textit{Hint}: the language 
\[L = \left\{ \alpha \in \{a,b\}^\ast 
\left| \
    \parbox{0.35\textwidth}
	        {$\alpha$ contains infinitely many $b$'s, and arbitrarily long blocks of $a$'s}
\right. \
\right\}\]
has finite index w.r.t. all three, yet is not $\omega$-regular.]
\end{exercise}



\begin{exercise}[co-Büchi automata]
A \emph{co-Büchi automaton} is an automaton
$\mathcal A=(Q,\Sigma,\delta,q_0,F)$ in which a run $\rho$
is accepting if
\[
\operatorname{Inf}(\rho)\subseteq F,
\]
i.e., if it visits states outside $F$ only finitely often.

The aim is to prove that every nondeterministic co-Büchi automaton is
equivalent to a deterministic co-Büchi automaton.
Adapt the subset construction, using as states pairs $(S,R)$ with
$R\subseteq S\subseteq Q$. The set $S$ tracks all states
reachable after reading the current prefix, whereas $R$
tracks those reachable by runs that have remained inside
$F$ since the last time $R$ was empty.

Initially, $(S,R)=(\{q_0\},\varnothing)$. On reading a
letter $a$, update
\[
S'=\delta(S,a),\qquad
R'=
\begin{cases}
\delta(S,a)\cap F,&R=\varnothing,\\
\delta(R,a)\cap F,&R\ne\varnothing.
\end{cases}
\]
Equip the resulting deterministic automaton with the
co-Büchi accepting set
\[
\{(S,R):R\ne\varnothing\}.
\]
Prove that it recognizes the same language as $\mathcal A$.
\end{exercise}
