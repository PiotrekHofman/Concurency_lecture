\begin{frame}{Linear programming + If formula.}
\Margines{
\DefinicjaProblemu{ A $r\times c$- integer matrix $M$ and a vector $\vec{y}\in \Z^{r}$ and a set of predicates of a form
$\vec{x}[\indexI]>0 \implies \vec{x}[\indexII]>0$.}{ If there is a vector $\vec{x}\in \Qplus^{c}$ such that
$M \cdot \vec{x}= \vec{y}$ and all predicates are satisfied?}
\begin{block}{Theorem }
The Linear programming + If formula problem is in \PTIME. 
\end{block}

\visible<2->{
\begin{block}{Proof}
\begin{enumerate}
 \item The set of solutions is convex.
 \item If for every $\indexI$ there is a solution such that $\vec{x}[\indexI]>0$ then there is a solution
 such that $\vec{x}[\indexII]>0$ for all $\indexII.$
\end{enumerate}
}
\end{block}


}\end{frame}

\begin{frame}{Continuous Petri Nets Reachability.}
\Margines{
\DefinicjaProblemu{ Two configurations $\Initial$ and $\Final$}{If there is a run form $\Initial$ to $\Final$ under continuous semantics.}
\begin{block}{A simpler variant of the problem.}
Suppose, that $$\forall_{\indexI}\ \left( \Initial[\indexI]>0 \text{ and } \Final[\indexI]>0\right).$$ 

$\Final$ is reachable from $\Initial$ iff
$$\Final-\Initial=\TransitionDelta \cdot \vec{x} \text{ where } \vec{x}\in \Qplus^\Dimension .$$
\end{block}
}
\end{frame}


\begin{frame}{Continuous Petri Nets Reachability.}
\Margines{
\begin{block}{Lemma}
$\Final$ is reachable from $\Initial$ if
\begin{enumerate}
 \item $$\Final-\Initial=\TransitionDelta \cdot \vec{x} \text{ where } \vec{x}\in \Qplus^\Dimension $$
 \item $$\vec{x}[\indexI]>0 \ \text{ and }\  \Pre[\indexII,\indexI]>0 \ \implies \ \Initial[\indexII]>0,$$
 \item $$\vec{x}[\indexI]>0 \ \text{ and }\  \Post[\indexII,\indexI]>0 \ \implies \ \Final[\indexII]>0.$$
\end{enumerate}

\end{block}
}
\end{frame}

\begin{frame}{Continuous Petri Nets Reachability.}
\Margines{
\begin{block}{Lemma}
$\Final$ is reachable from $\Initial$ if
\begin{enumerate}
 \item $\Final-\Initial=\TransitionDelta \cdot \vec{x} \text{ where } \vec{x}\in \Qplus^\Dimension $
 \item $\vec{x}[\indexI]>0 \ \text{ and }\  \Pre[\indexII,\indexI]>0 \ \implies \ \Initial[\indexII]>0,$
 \item $\vec{x}[\indexI]>0 \ \text{ and }\  \Post[\indexII,\indexI]>0 \ \implies \ \Final[\indexII]>0.$
\end{enumerate}
\end{block}
\begin{block}{Theorem}
$\Final$ is reachable from $\Initial$ iff there are two configurations $\Initial'$ and $\Final'$ such that
\begin{enumerate}
 \item there is a run form $\Initial$ to $\Initial'$ that is using at most $\Dimension$ steps.
 \item there is a run form $\Final'$ to $\Final$ that is using at most $\Dimension$ steps.
 \item There is a run form $\Initial'$ to $\Final'$ due to Lemma. 
\end{enumerate}
\end{block}
}
\end{frame}




\begin{frame}{Translation to a formula (linear + If).}
\Margines{
\begin{block}{Lemma}
 For a given Petri net $\NetI$ and two configurations $\initial$ and $\final$
 in \PTIME\ one can compute a formula (linear programming + if) such that it is satisfiable if and only if
 $\final$ is continuously reachable from $\initial$ in the net $\NetI.$
\end{block}

We use:
\begin{block}{Theorem}
$\Final$ is reachable from $\Initial$ iff there are two configurations $\Initial'$ and $\Final'$ such that
\begin{enumerate}
 \item there is a run form $\Initial$ to $\Initial'$ that is using at most $\Dimension$ steps.
 \item there is a run form $\Final'$ to $\Final$ that is using at most $\Dimension$ steps.
 \item There is a run form $\Initial'$ to $\Final'$ due to Lemma. 
\end{enumerate}
\end{block}
}
\end{frame}
