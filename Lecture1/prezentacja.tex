\begin{frame}{Labelled Transition Systems (LTS)}
\Margines{


\visible<1->{\begin{block}{Definition}\begin{scriptsize}
 A Labelled Transition System (LTS) is a 3-tuple $T = (S, A, \rightarrow)$ consisting of:
\begin{enumerate}
 \item a set of states $S$ (possibly infinite),
 \item a set of actions (or labels) $A$,
 \item a transition relation $\rightarrow \subseteq S \times A \times S$.
 \end{enumerate}
 We write $s \xrightarrow{a} s'$ to denote $(s, a, s') \in \rightarrow$.
\end{scriptsize}
\end{block}}

\visible<2->{\begin{block}{Definition}\begin{scriptsize}
A \emph{run} in an LTS is a sequence of states and actions:
$$s_1 \xrightarrow{a_1} s_2 \xrightarrow{a_2} s_3 \xrightarrow{a_3} \ldots$$
such that $(s_i, a_i, s_{i+1}) \in \rightarrow$ for all $i \geq 1$.

A \emph{trace} is the sequence of actions observable in a run:
$$a_1, a_2, a_3, \ldots$$
The set of all traces of all possible infinite runs of an LTS $T$ is called \emph{Traces} of $T$ and denoted $\trace{T}$.
\end{scriptsize}

\end{block}}

}

\end{frame}

\begin{frame}{Kripke structures}
\Margines{
Let $\AProp$ be a set of atomic propositions, i.e. boolean expressions over variables,
constants and predicate symbols.
\visible<2->{\begin{block}{Definition}
 A Kripke structure over $\AProp$ is a 4-tuple $M = (S, I, R, L)$ consisting of:
\begin{enumerate}
 \item a finite set of states $S,$
 \item a set of initial states $I \subseteq S,$
 \item a transition relation $R \subseteq S \times S$ such that $R$ is left-total, i.e., $\forall_{s \in S} \exists_{s'\in S}$ such that $(s,s')\in R$,
 \item a labeling (or interpretation) function $L: S\rightarrow 2^{\AProp}$.
 \end{enumerate}
 \end{block}}
\visible<3->{\begin{block}{Definition}
  By a \emph{run} we mean a sequence of states interleaved with transitions, $s_1,t_1,s_2,t_2,s_3\ldots$ such that $(s_i,s_{i+1})=t_{i}$.
 \end{block}
 }}
\end{frame}


\begin{frame}{Traces}
\Margines{
\begin{definition}
For a given run we define a \emph{trace} as follows:
$$\trace{s_1,t_1,s_2,t_2,s_3\ldots}= L(s_1),L(s_2),L(s_3)\ldots.$$
  A set of traces of all possible infinite runs of a given Kripke structure $S$ is called \emph{Traces} of $S$ and denoted $\trace{S}.$
\end{definition}

\begin{frame}{Pros and cons of LTS and Kripke structures}
\Margines{
\begin{block}{Pros}
\begin{itemize}
 \item<1-> Are very simple and intuitive.
 \item<1-> Can be used to model a system.
 \item <1-> Useful for defining semantics of other modelling formalisms.
\end{itemize}
\end{block}
\begin{block}{Cons}
\begin{itemize}
 \item<2-> Are not very convenient to write.
 \item<2-> Are not very convenient to read.
 \item<2-> Models are very big.
 \item<3-> They do not explicitly show imperfect information.
 \end{itemize}
\end{block}
}
\visible<4->{
But for a few first lectures we will use them as a model of systems.
}
\end{frame}

\begin{frame}{Automaton}
\Margines{
Traces are like a language, so we can try with automata.

Let $\Sigma$ be a set of letters (a finite alphabet).

\begin{block}{Definition}
Automaton is an ordered 5-tuple $A=(S,I,F, R, L)$ where:
\begin{enumerate}
 \item $S$ is a finite set of states,
 \item $I$ is a set of initial states, $I \subseteq S,$
 \item $F$ is a set of accepting states, $F \subseteq S,$
 \item $R$ is a transition relation $R \subseteq S \times S$,
 \item $L$ is a labeling (or interpretation) function $L: R\rightarrow \Sigma$.
 \end{enumerate}
\end{block}

\alt<3>{
\begin{block}{\red{Problem}}
 Traces are infinite words and words accepted by a nondeterministic automaton are finite.
\end{block}}
{\visible<2->{\begin{block}{Definition}
A language of an automaton $A$ is a set of words $\subseteq \Sigma^*$ such that they can be read along
a path from an initial state to a final state.
\end{block}
}}
}
\end{frame}

\begin{frame}{Characterisations of languages of infinite words}
 \Margines{
 Let's try to define automata on infinite words.
 \begin{itemize}
  \item<2-> Attempt 1: concatenate word with $\Sigma^\omega$.
  \item<3-> Attempt 2: an infinite word is accepted if from some moment a run stays only in accepting states.
  \item<4-> Attempt 3: Buchi\ automaton, word is accepted if it visits accepting states infinitely often.
  \item<5-> Attempt 4: A generalised Buchi\ automaton.
 \end{itemize}
\visible<5->{\vspace{-0.3cm}\begin{block}{Definition}
A generalised \Buchi\ automaton is an ordered 5-tuple $A=(S,I,F, R, L)$ where:
\begin{enumerate}
 \only<5>{\item $S$ is a finite set of states,
 \item $I$ is a set of initial states, $I \subseteq S,$}
 \item $F$ is a finite set of sets $\{F_1\ldots F_k\}$ of accepting states, $F_i \subseteq S,$
 \only<5>{\item $R$ is a transition relation $R \subseteq S \times S$,
 \item $L$ is a labelling (or interpretation) function $L: R\rightarrow \Sigma$.}
 \end{enumerate}
\end{block}
}
\visible<7>{\begin{block}{Definition}
A word is accepted if it visits infinitely often states in $F_i$ for every $0<i\leq k.$
\end{block}}
}
\end{frame}



\begin{frame}{\Buchi\ languages}
\Margines{

\vspace{-0.cm}

\begin{lemma}
 For a given Kripke structure $S$ there is a \Buchi\ automaton $A$ such that:
 $$\lang{A}=\trace{S}.$$
\end{lemma}

\visible<2->{
\begin{block}{Lemma}
 Sets of languages accepted by \Buchi\ automata and generalised \Buchi\ automata are the same.
\end{block}}

\visible<3->{\begin{block}{Lemma}
 \Buchi\ languages are closed under:
 \begin{enumerate}
  \item Union,
  \item Intersection.
  \item Complement. (We will not do this)
  \item Determinisation does not work. (We need to extend the model)
 \end{enumerate}
\end{block}}
}
